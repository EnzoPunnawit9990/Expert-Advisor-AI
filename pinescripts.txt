//@version=6
indicator("ORDER BLOCK QUALITY",
     shorttitle = "HQ-OB-Q",
     overlay = true,
     max_boxes_count = 500,
     max_labels_count = 500,
     max_lines_count = 500)

// =====================================================================================
// Flow: OB Origin -> ATR Strength -> BOS + FVG -> Unmitigated -> OB Width/ATR -> Quality Score
// ตัดสินใจเฉพาะแท่งที่ปิดแล้ว (barstate.isconfirmed) ไม่ Repaint
// =====================================================================================

// ============================================================
// INPUTS
// ============================================================
swingLength       = input.int(5, "Swing Length", minval = 1, group = "Structure (BOS)")
maxCandidateAge   = input.int(50, "Max Bars Waiting for BOS Confirmation", minval = 5, maxval = 300, group = "Structure (BOS)")

atrLength         = input.int(14, "ATR Length", minval = 1, group = "Displacement")
minDispStrength   = input.float(1.0, "Minimum Displacement Strength (x ATR)", minval = 0.1, step = 0.1, group = "Displacement")

maxObToFvgDistance = input.int(3, "Max OB to FVG Distance (bars)", minval = 0, maxval = 20, group = "Imbalance (FVG)")

mitigationMethod  = input.string("Candle Close Inside OB", "Mitigation Method",
     options = ["Wick Touch", "Candle Close Inside OB", "Full Body Close Through OB"], group = "Mitigation")

maxActiveOB       = input.int(20, "Maximum Active OB", minval = 1, maxval = 100, group = "Display")
showMitigatedOB   = input.bool(false, "Show Mitigated OB", group = "Display")

// ---------- ATR Strength ----------
minAtrShow  = input.float(3.0, "Minimum ATR Strength to Display (x ATR)", minval = 0.0, step = 0.1, group = "ATR Strength")
atrThr1     = input.float(1.0, "Strong Threshold (x ATR)", minval = 0.1, step = 0.1, group = "ATR Strength")
atrThr2     = input.float(3.0, "Very Strong Threshold (x ATR)", minval = 0.1, step = 0.1, group = "ATR Strength")
atrThr3     = input.float(5.0, "Extreme Threshold (x ATR)", minval = 0.1, step = 0.1, group = "ATR Strength")
atrCol1     = input.color(color.new(#2962FF, 0), "Strong Color", group = "ATR Strength")
atrCol2     = input.color(color.new(#0D2A8C, 0), "Very Strong Color", group = "ATR Strength")
atrCol3     = input.color(color.new(#D500F9, 0), "Extreme Color", group = "ATR Strength")

// ---------- Quality Scoring ----------
minQuality      = input.float(0.0, "Minimum Quality Score to Display (0-100)", minval = 0.0, maxval = 100.0, step = 5.0, group = "OB Quality")
idealWidthATR   = input.float(1.0, "Ideal OB Width (x ATR) - full score", minval = 0.1, step = 0.1, group = "OB Quality")
maxWidthATR     = input.float(3.0, "Max OB Width (x ATR) - reject if wider", minval = 0.2, step = 0.1, group = "OB Quality")
bosIdealATR     = input.float(0.5, "Ideal BOS Break Distance (x ATR)", minval = 0.05, step = 0.05, group = "OB Quality")
fvgIdealATR     = input.float(0.3, "Ideal FVG Size (x ATR)", minval = 0.05, step = 0.05, group = "OB Quality")
wAtr            = input.float(30.0, "Weight: ATR Strength", minval = 0.0, group = "OB Quality")
wBosFvg         = input.float(25.0, "Weight: BOS + FVG", minval = 0.0, group = "OB Quality")
wFresh          = input.float(20.0, "Weight: Unmitigated (Freshness)", minval = 0.0, group = "OB Quality")
wWidth          = input.float(25.0, "Weight: OB Width", minval = 0.0, group = "OB Quality")

// ---------- Style ----------
colorMode       = input.string("Quality Grade", "Box Color By", options = ["Quality Grade", "ATR Strength"], group = "Style")
showLabel       = input.bool(true, "Show Quality Label", group = "Style")
fillTransp      = input.int(80, "Box Fill Transparency", minval = 0, maxval = 100, group = "Style")
atrTextColor    = input.color(color.white, "Text Color", group = "Style")
atrTextSize     = input.string("Small", "Text Size", options = ["Tiny", "Small", "Normal", "Large"], group = "Style")
gradeColAP      = input.color(#00C853, "Grade A+ Color", group = "Style")
gradeColA       = input.color(#2962FF, "Grade A Color", group = "Style")
gradeColB       = input.color(#FFB300, "Grade B Color", group = "Style")
gradeColC       = input.color(#9E9E9E, "Grade C Color", group = "Style")

bullBgColor   = color.new(color.teal, 80)
bearBgColor   = color.new(color.red, 80)
bullBorderCol = color.new(color.teal, 0)
bearBorderCol = color.new(color.red, 0)
mitigatedBg   = color.new(color.gray, 90)
mitigatedBord = color.new(color.gray, 60)

// ============================================================
// SWING / DISPLACEMENT
// ============================================================
float ph = ta.pivothigh(swingLength, swingLength)
float pl = ta.pivotlow(swingLength, swingLength)

var float structHighLevel = na
var float structLowLevel  = na
if not na(ph)
    structHighLevel := ph
if not na(pl)
    structLowLevel := pl

float atrVal      = ta.atr(atrLength)
float dispStrength = atrVal > 0 ? math.abs(close - open) / atrVal : 0.0
bool  isBullDisp   = close > open and dispStrength >= minDispStrength
bool  isBearDisp   = close < open and dispStrength >= minDispStrength

findBearishOrigin() =>
    int idx = na
    for i = 1 to 10
        if close[i] < open[i]
            idx := i
            break
    idx

findBullishOrigin() =>
    int idx = na
    for i = 1 to 10
        if close[i] > open[i]
            idx := i
            break
    idx

// ============================================================
// ARRAYS
// ============================================================
// Pending candidates (รอ BOS)
var array<float> pBullHigh     = array.new<float>()
var array<float> pBullLow      = array.new<float>()
var array<int>   pBullObBar    = array.new<int>()
var array<int>   pBullDispBar  = array.new<int>()
var array<float> pBullSwingRef = array.new<float>()
var array<float> pBullFvg      = array.new<float>()   // ขนาด FVG (x ATR)

var array<float> pBearHigh     = array.new<float>()
var array<float> pBearLow      = array.new<float>()
var array<int>   pBearObBar    = array.new<int>()
var array<int>   pBearDispBar  = array.new<int>()
var array<float> pBearSwingRef = array.new<float>()
var array<float> pBearFvg      = array.new<float>()

// Confirmed OB
var array<box>   obBox    = array.new<box>()
var array<int>   obDir    = array.new<int>()
var array<float> obHighA  = array.new<float>()
var array<float> obLowA   = array.new<float>()
var array<bool>  obMitig  = array.new<bool>()
var array<float> obStrX   = array.new<float>()   // ATR Strength (x)
var array<float> obBosX   = array.new<float>()   // ระยะ BOS (x ATR)
var array<float> obFvgX   = array.new<float>()   // ขนาด FVG (x ATR)
var array<float> obWidX   = array.new<float>()   // Width (x ATR)
var array<int>   obTouch  = array.new<int>()     // จำนวนแท่งที่แตะโซนแต่ยังไม่ Mitigate
var array<float> obScore  = array.new<float>()   // Quality Score 0-100

// ============================================================
// HELPERS
// ============================================================
strengthTier(float s) =>
    int t = 0
    if s >= atrThr3
        t := 3
    else if s >= atrThr2
        t := 2
    else if s >= atrThr1
        t := 1
    t

txtSize() =>
    switch atrTextSize
        "Tiny"  => size.tiny
        "Small" => size.small
        "Large" => size.large
        => size.normal

gradeOf(float s) =>
    s >= 85 ? "A+" : s >= 70 ? "A" : s >= 55 ? "B" : "C"

gradeColor(float s) =>
    s >= 85 ? gradeColAP : s >= 70 ? gradeColA : s >= 55 ? gradeColB : gradeColC

// คำนวณ Quality Score 0-100
calcScore(float strengthX, float bosX, float fvgX, float widthX, int touches) =>
    float sAtr   = math.min(strengthX / atrThr3, 1.0)
    float sBos   = math.min(bosX / bosIdealATR, 1.0)
    float sFvg   = math.min(fvgX / fvgIdealATR, 1.0)
    float sBF    = 0.4 * sBos + 0.6 * sFvg
    float sFresh = touches == 0 ? 1.0 : touches == 1 ? 0.5 : 0.0
    float wRange = math.max(maxWidthATR - idealWidthATR, 0.01)
    float sW     = widthX <= idealWidthATR ? 1.0 : widthX >= maxWidthATR ? 0.0 : 1.0 - (widthX - idealWidthATR) / wRange
    float wSum   = wAtr + wBosFvg + wFresh + wWidth
    wSum > 0 ? 100.0 * (sAtr * wAtr + sBF * wBosFvg + sFresh * wFresh + sW * wWidth) / wSum : 0.0

// ตั้งสี + ข้อความของกล่องตามคะแนน
applyStyle(int i) =>
    box   bx  = array.get(obBox, i)
    int   dir = array.get(obDir, i)
    float sc  = array.get(obScore, i)
    float sx  = array.get(obStrX, i)
    float wx  = array.get(obWidX, i)
    color bdC = dir == 1 ? bullBorderCol : bearBorderCol
    color bgC = dir == 1 ? bullBgColor : bearBgColor
    if colorMode == "Quality Grade"
        color gc = gradeColor(sc)
        bdC := gc
        bgC := color.new(gc, fillTransp)
    else
        int t = strengthTier(sx)
        if t > 0
            color tc = t == 3 ? atrCol3 : t == 2 ? atrCol2 : atrCol1
            bdC := tc
            bgC := color.new(tc, fillTransp)
    box.set_border_color(bx, bdC)
    box.set_bgcolor(bx, bgC)
    string txt = showLabel ? gradeOf(sc) + " " + str.tostring(sc, "#") + "\nATR " + str.tostring(sx, "#.0") + "x | W " + str.tostring(wx, "#.0") + "x" : ""
    box.set_text(bx, txt)
    box.set_text_color(bx, atrTextColor)
    box.set_text_size(bx, txtSize())

removeOB(int i) =>
    box.delete(array.get(obBox, i))
    array.remove(obBox, i)
    array.remove(obDir, i)
    array.remove(obHighA, i)
    array.remove(obLowA, i)
    array.remove(obMitig, i)
    array.remove(obStrX, i)
    array.remove(obBosX, i)
    array.remove(obFvgX, i)
    array.remove(obWidX, i)
    array.remove(obTouch, i)
    array.remove(obScore, i)

// สร้าง OB เมื่อผ่านเกณฑ์ทั้งหมด (ATR Strength, Width, Quality)
addOB(int dir, int obBarIdx, float obHighV, float obLowV, float strX, float bosX, float fvgX, int touches) =>
    float widthX = atrVal > 0 ? (obHighV - obLowV) / atrVal : 0.0
    float score  = calcScore(strX, bosX, fvgX, widthX, touches)
    if strX >= minAtrShow and widthX <= maxWidthATR and score >= minQuality
        box bx = box.new(left = obBarIdx, top = obHighV, right = bar_index, bottom = obLowV,
             border_color = bullBorderCol, bgcolor = bullBgColor, extend = extend.none,
             text = "", text_color = atrTextColor, text_size = txtSize(),
             text_halign = text.align_center, text_valign = text.align_center)
        array.push(obBox, bx)
        array.push(obDir, dir)
        array.push(obHighA, obHighV)
        array.push(obLowA, obLowV)
        array.push(obMitig, false)
        array.push(obStrX, strX)
        array.push(obBosX, bosX)
        array.push(obFvgX, fvgX)
        array.push(obWidX, widthX)
        array.push(obTouch, touches)
        array.push(obScore, score)
        applyStyle(array.size(obBox) - 1)

// ============================================================
// MAIN
// ============================================================
if barstate.isconfirmed

    // -------- STEP 1: Displacement + FVG -> Pending Candidate --------
    if isBullDisp and not na(structHighLevel)
        int originIdx = findBearishOrigin()
        if not na(originIdx)
            float fvgMax = 0.0
            for j = 0 to maxObToFvgDistance
                float g = low[j] - high[j + 2]
                if g > 0
                    fvgMax := math.max(fvgMax, g)
            if fvgMax > 0
                array.push(pBullHigh, high[originIdx])
                array.push(pBullLow, low[originIdx])
                array.push(pBullObBar, bar_index - originIdx)
                array.push(pBullDispBar, bar_index)
                array.push(pBullSwingRef, structHighLevel)
                array.push(pBullFvg, atrVal > 0 ? fvgMax / atrVal : 0.0)

    if isBearDisp and not na(structLowLevel)
        int originIdx = findBullishOrigin()
        if not na(originIdx)
            float fvgMax = 0.0
            for j = 0 to maxObToFvgDistance
                float g = low[j + 2] - high[j]
                if g > 0
                    fvgMax := math.max(fvgMax, g)
            if fvgMax > 0
                array.push(pBearHigh, high[originIdx])
                array.push(pBearLow, low[originIdx])
                array.push(pBearObBar, bar_index - originIdx)
                array.push(pBearDispBar, bar_index)
                array.push(pBearSwingRef, structLowLevel)
                array.push(pBearFvg, atrVal > 0 ? fvgMax / atrVal : 0.0)

    // -------- STEP 2: ตรวจ BOS (Bullish) --------
    if array.size(pBullHigh) > 0
        for k = array.size(pBullHigh) - 1 to 0
            float swingRef = array.get(pBullSwingRef, k)
            int   dispBar  = array.get(pBullDispBar, k)
            int   obBarIdx = array.get(pBullObBar, k)
            float obLowV   = array.get(pBullLow, k)
            float obHighV  = array.get(pBullHigh, k)
            float fvgX     = array.get(pBullFvg, k)
            bool  drop     = false
            if bar_index - dispBar > maxCandidateAge
                drop := true
            else if close < obLowV
                drop := true
            else if close > swingRef and bar_index > dispBar
                // ATR Strength
                float extremeHigh = high
                int   barsBack    = bar_index - obBarIdx
                for q = 0 to barsBack
                    extremeHigh := math.max(extremeHigh, high[q])
                float strX = atrVal > 0 ? (extremeHigh - obLowV) / atrVal : 0.0
                // Unmitigated: นับแท่งที่แตะโซนหลัง Displacement จนถึงตอนยืนยัน
                int touches = 0
                for q = 0 to bar_index - dispBar - 1
                    if low[q] <= obHighV
                        touches += 1
                float bosX = atrVal > 0 ? (close - swingRef) / atrVal : 0.0
                addOB(1, obBarIdx, obHighV, obLowV, strX, bosX, fvgX, touches)
                drop := true
            if drop
                array.remove(pBullHigh, k)
                array.remove(pBullLow, k)
                array.remove(pBullObBar, k)
                array.remove(pBullDispBar, k)
                array.remove(pBullSwingRef, k)
                array.remove(pBullFvg, k)

    // -------- STEP 2: ตรวจ BOS (Bearish) --------
    if array.size(pBearHigh) > 0
        for k = array.size(pBearHigh) - 1 to 0
            float swingRef = array.get(pBearSwingRef, k)
            int   dispBar  = array.get(pBearDispBar, k)
            int   obBarIdx = array.get(pBearObBar, k)
            float obLowV   = array.get(pBearLow, k)
            float obHighV  = array.get(pBearHigh, k)
            float fvgX     = array.get(pBearFvg, k)
            bool  drop     = false
            if bar_index - dispBar > maxCandidateAge
                drop := true
            else if close > obHighV
                drop := true
            else if close < swingRef and bar_index > dispBar
                float extremeLow = low
                int   barsBack   = bar_index - obBarIdx
                for q = 0 to barsBack
                    extremeLow := math.min(extremeLow, low[q])
                float strX = atrVal > 0 ? (obHighV - extremeLow) / atrVal : 0.0
                int touches = 0
                for q = 0 to bar_index - dispBar - 1
                    if high[q] >= obLowV
                        touches += 1
                float bosX = atrVal > 0 ? (swingRef - close) / atrVal : 0.0
                addOB(-1, obBarIdx, obHighV, obLowV, strX, bosX, fvgX, touches)
                drop := true
            if drop
                array.remove(pBearHigh, k)
                array.remove(pBearLow, k)
                array.remove(pBearObBar, k)
                array.remove(pBearDispBar, k)
                array.remove(pBearSwingRef, k)
                array.remove(pBearFvg, k)

    // -------- STEP 3: ยืด Zone / ตรวจ Mitigation / อัปเดต Quality --------
    if array.size(obBox) > 0
        for m = array.size(obBox) - 1 to 0
            box   bx  = array.get(obBox, m)
            int   dir = array.get(obDir, m)
            float obH = array.get(obHighA, m)
            float obL = array.get(obLowA, m)
            bool  mit = array.get(obMitig, m)

            if not mit
                bool isMit = false
                if dir == 1
                    if mitigationMethod == "Wick Touch"
                        isMit := low <= obH
                    else if mitigationMethod == "Candle Close Inside OB"
                        isMit := close <= obH and close >= obL
                    else
                        isMit := open > obH and close < obL
                else
                    if mitigationMethod == "Wick Touch"
                        isMit := high >= obL
                    else if mitigationMethod == "Candle Close Inside OB"
                        isMit := close >= obL and close <= obH
                    else
                        isMit := open < obL and close > obH

                if isMit
                    array.set(obMitig, m, true)
                    if showMitigatedOB
                        box.set_bgcolor(bx, mitigatedBg)
                        box.set_border_color(bx, mitigatedBord)
                        box.set_text(bx, "")
                        box.set_right(bx, bar_index)
                    else
                        removeOB(m)
                else
                    box.set_right(bx, bar_index)
                    // แตะโซนแต่ยังไม่ Mitigate -> ลดคะแนน Freshness
                    bool touched = dir == 1 ? low <= obH : high >= obL
                    if touched
                        array.set(obTouch, m, array.get(obTouch, m) + 1)
                        float ns = calcScore(array.get(obStrX, m), array.get(obBosX, m),
                             array.get(obFvgX, m), array.get(obWidX, m), array.get(obTouch, m))
                        array.set(obScore, m, ns)
                        applyStyle(m)
            else if showMitigatedOB
                box.set_right(bx, bar_index)

    // -------- STEP 4: จำกัดจำนวน Active OB --------
    int activeCount = 0
    if array.size(obMitig) > 0
        for n = 0 to array.size(obMitig) - 1
            if not array.get(obMitig, n)
                activeCount += 1

    while activeCount > maxActiveOB
        int removeIdx = na
        for n = 0 to array.size(obMitig) - 1
            if not array.get(obMitig, n)
                removeIdx := n
                break
        if not na(removeIdx)
            removeOB(removeIdx)
            activeCount -= 1
        else
            break
